-- 1. Tabla de vehículos autorizados y fichas
create table public.vehiculos (
    id bigint generated by default as identity primary key,
    patente text unique not null,
    marca text not null,
    modelo text not null,
    color text not null,
    propietario text not null,
    unidad text not null,
    presion rueda 1 text not null,
    presión rueda 2 text not null,
    presión rueda 3 text not null,
    presión rueda 4 text not null,
    nivel de aceite text not null,
    nivel de combustible text not null,
    voltaje de bateria text not null,
    luces delanteras text not null,
    luces traseras text not null,
    balizas text not null,
    autorizado boolean default true not null,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Tabla de registros de acceso (historial con referencia a foto)
create table public.registros_acceso (
    id bigint generated by default as identity primary key,
    patente text not null,
    movimiento text not null, -- 'ENTRADA' o 'SALIDA'
    imagen_url text, -- URL de la evidencia en Supabase Storage
    fecha_hora timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 3. Habilitar seguridad a nivel de fila (RLS)
alter table public.vehiculos enable row level security;
alter table public.registros_acceso enable row level security;

create policy "Acceso publico vehiculos" on public.vehiculos for all using (true) with check (true);
create policy "Acceso publico registros" on public.registros_acceso for all using (true) with check (true);
